<script>
(function(){
  const STORAGE_KEY='disciplineCalendarData_v_continuous';
  const START_YEAR=2025, END_YEAR=2035;
  // 初始 maxCount（可以被用户修改）
  let maxCount = 20;

  // 颜色由白 -> 绿 -> 蓝 -> 紫
  const BELT_COLORS = ['#ffffff','#66bb6a','#42a5f5','#7e57c2'];

  let currentDate=new Date();
  let viewMode='year';

  const grid=document.getElementById('dc-grid');
  const viewSelect=document.getElementById('view-mode');
  const btnPrev=document.getElementById('btn-prev');
  const btnNext=document.getElementById('btn-next');
  const btnCheckin=document.getElementById('btn-checkin');
  const btnExport=document.getElementById('btn-export');
  const btnImport=document.getElementById('btn-import');
  const fileImport=document.getElementById('file-import');
  const statTotal=document.getElementById('stat-total');
  const statStreak=document.getElementById('stat-streak');
  const statLast=document.getElementById('stat-last');
  const statCurrent=document.getElementById('stat-current');
  const maxCountInput=document.getElementById('max-count-input');

  const lg0=document.getElementById('lg-0');
  const lg1=document.getElementById('lg-1');
  const lg2=document.getElementById('lg-2');
  const lg3=document.getElementById('lg-3');
  const lg4=document.getElementById('lg-4');
  const lg5=document.getElementById('lg-5');

  function hexToRgb(hex){
    const h = hex.replace('#','');
    return [parseInt(h.substring(0,2),16), parseInt(h.substring(2,4),16), parseInt(h.substring(4,6),16)];
  }
  function rgbToHex(r,g,b){
    const toHex=(n)=>('0'+n.toString(16)).slice(-2);
    return '#'+toHex(r)+toHex(g)+toHex(b);
  }

  function interpColor(c1,c2,t){
    return [Math.round(c1[0]+(c2[0]-c1[0])*t), Math.round(c1[1]+(c2[1]-c1[1])*t), Math.round(c1[2]+(c2[2]-c1[2])*t)];
  }

  // 连续颜色映射：0 -> BELT_COLORS[0], maxCount -> BELT_COLORS[last]
  function colorScale(count){
    const capped = Math.max(0, Math.min(count, maxCount));
    const ratio = maxCount > 0 ? (capped / maxCount) : 0; // 0..1
    const stops = BELT_COLORS.map(hexToRgb);
    const pos = ratio * (stops.length - 1);
    const idx = Math.floor(pos);
    const t = pos - idx;
    if(idx >= stops.length - 1) return `rgb(${stops[stops.length-1].join(',')})`;
    const c = interpColor(stops[idx], stops[idx+1], t);
    return `rgb(${c.join(',')})`;
  }

  function loadData(){try{return JSON.parse(localStorage.getItem(STORAGE_KEY)||'{}');}catch(e){return{}}}
  function saveData(data){localStorage.setItem(STORAGE_KEY,JSON.stringify(data));}
  function dateKey(d){return d.toISOString().slice(0,10);}

  function getDates(){
    const dates=[];
    if(viewMode==='year'){
      const year=currentDate.getFullYear();
      const start=new Date(year,0,1), end=new Date(year+1,0,1);
      for(let d=new Date(start);d<end;d.setDate(d.getDate()+1))dates.push(new Date(d));
    } else if(viewMode==='month'){
      const y=currentDate.getFullYear(), m=currentDate.getMonth();
      const start=new Date(y,m,1), end=new Date(y,m+1,1);
      for(let d=new Date(start);d<end;d.setDate(d.getDate()+1))dates.push(new Date(d));
    } else if(viewMode==='week'){
      const day=currentDate.getDay();
      const start=new Date(currentDate); start.setDate(start.getDate()-day);
      for(let i=0;i<7;i++){const d=new Date(start); d.setDate(start.getDate()+i);dates.push(d);}
    }
    return dates;
  }

  function render(){
    // 更新图例（均匀取 6 个点显示）
    const legendSteps = [0, Math.round(maxCount*0.2), Math.round(maxCount*0.4), Math.round(maxCount*0.6), Math.round(maxCount*0.8), maxCount];
    [lg0,lg1,lg2,lg3,lg4,lg5].forEach((el,i)=>{ el.style.background=colorScale(legendSteps[i]); el.title=`${legendSteps[i]}`; });

    grid.innerHTML='';
    const dates=getDates();
    const data=loadData();
    dates.forEach(day=>{
      const cell=document.createElement('div');
      cell.className='dc-cell';
      const k=dateKey(day);
      const cnt=data[k]||0;
      cell.dataset.date=k;
      cell.style.background = colorScale(cnt);
      cell.title=`${k}\ndiscipline: ${cnt}`;
      cell.tabIndex=0;

      let touchTimer=null;
      const inc=(e)=>{e&&e.preventDefault(); updateCount(cell,+1);};
      const dec=(e)=>{e&&e.preventDefault(); updateCount(cell,-1);};
      cell.addEventListener('click',()=>inc());
      cell.addEventListener('keydown',(e)=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();inc();} if(e.key==='-'){e.preventDefault();dec();}});
      cell.addEventListener('touchstart',()=>{touchTimer=setTimeout(()=>dec(),600);},{passive:true});
      cell.addEventListener('touchend',()=>{if(touchTimer)clearTimeout(touchTimer);});
      cell.addEventListener('contextmenu',(e)=>{e.preventDefault(); dec();});
      grid.appendChild(cell);
    });
    updateStats();
  }

  function updateCount(cell,delta){
    const k=cell.dataset.date;
    if(!k)return;
    const data=loadData();
    data[k]=Math.max(0,(data[k]||0)+delta);
    saveData(data);
    render();
  }

  function updateStats(){
    const data=loadData();
    statTotal.textContent=Object.values(data).reduce((a,b)=>a+(b||0),0);
    const today=new Date();
    let streak=0;
    for(let i=0;i<1000;i++){
      const d=new Date(today.getFullYear(),today.getMonth(),today.getDate()-i);
      if((data[dateKey(d)]||0)>0) streak++; else break;
    }
    statStreak.textContent=streak;
    const keys=Object.keys(data).filter(k=>data[k]>0).sort();
    statLast.textContent=keys.length?keys[keys.length-1]:'—';
    if(viewMode==='year') statCurrent.textContent=currentDate.getFullYear();
    else if(viewMode==='month') statCurrent.textContent=`${currentDate.getFullYear()}-${currentDate.getMonth()+1}`;
    else if(viewMode==='week') {
      const start=new Date(currentDate); start.setDate(start.getDate()-start.getDay());
      const end=new Date(start); end.setDate(start.getDate()+6);
      statCurrent.textContent=`${start.toISOString().slice(0,10)} ~ ${end.toISOString().slice(0,10)}`;
    }
  }

  viewSelect.addEventListener('change',(e)=>{viewMode=e.target.value; render();});
  btnPrev.addEventListener('click',()=>{
    if(viewMode==='year'){ if(currentDate.getFullYear()>START_YEAR) currentDate.setFullYear(currentDate.getFullYear()-1);} 
    else if(viewMode==='month'){ currentDate.setMonth(currentDate.getMonth()-1);} 
    else if(viewMode==='week'){ currentDate.setDate(currentDate.getDate()-7);} 
    render();
  });
  btnNext.addEventListener('click',()=>{
    if(viewMode==='year'){ if(currentDate.getFullYear()<END_YEAR) currentDate.setFullYear(currentDate.getFullYear()+1);} 
    else if(viewMode==='month'){ currentDate.setMonth(currentDate.getMonth()+1);} 
    else if(viewMode==='week'){ currentDate.setDate(currentDate.getDate()+7);} 
    render();
  });

  btnCheckin.addEventListener('click',()=>{
    const k=dateKey(new Date());
    const data=loadData();
    data[k]=(data[k]||0)+1;
    saveData(data);
    render();
  });

  btnExport.addEventListener('click',()=>{
    const data=loadData();
    const blob=new Blob([JSON.stringify(data,null,2)],{type:'application/json'});
    const url=URL.createObjectURL(blob);
    const a=document.createElement('a'); a.href=url;a.download=`discipline-calendar-${new Date().toISOString().slice(0,10)}.json`; document.body.appendChild(a); a.click(); a.remove(); URL.revokeObjectURL(url);
  });

  btnImport.addEventListener('click',()=>fileImport.click());
  fileImport.addEventListener('change',(e)=>{
    const f=e.target.files && e.target.files[0]; if(!f)return;
    const reader=new FileReader();
    reader.onload=(ev)=>{
      try{
        const j=JSON.parse(ev.target.result);
        const cur=loadData();
        saveData(Object.assign({},cur,j));
        render();
        alert('导入成功');
      }catch(err){alert('导入失败：文件不是有效 JSON');}
    };
    reader.readAsText(f,'utf-8'); e.target.value='';
  });

  // 当用户修改 maxCount 时更新并重新渲染
  maxCountInput.addEventListener('change',(e)=>{
    let v = parseInt(e.target.value,10);
    if(!isFinite(v) || v < 1) v = 1;
    maxCount = v;
    e.target.value = v;
    render();
  });

  // 如果你想把 maxCount 存到 localStorage 保持跨次打开，可以在这里读取/写入
  (function loadMaxCount(){
    try{
      const m = parseInt(localStorage.getItem(STORAGE_KEY+"_maxCount"),10);
      if(isFinite(m) && m>=1){ maxCount = m; maxCountInput.value = m; }
    }catch(e){}
  })();

  // 在用户修改后保存 maxCount（避免频繁写入，这里在离开页面时保存）
  window.addEventListener('beforeunload',()=>{ try{ localStorage.setItem(STORAGE_KEY+"_maxCount", String(maxCount)); }catch(e){} });

  render();
})();
</script>
